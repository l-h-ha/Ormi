cmake_minimum_required(VERSION 3.18)

project(ormi_cuda_package LANGUAGES CXX)

set(ORMI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ORMI_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(ORMI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ORMI_BUILD_PYTHON ON CACHE BOOL "" FORCE)

set(ORMI_ENABLE_CUDA ON CACHE BOOL "" FORCE)

add_subdirectory(
    "${CMAKE_CURRENT_SOURCE_DIR}/../ormi-core"
    "${CMAKE_CURRENT_BINARY_DIR}/ormi-core"
)
